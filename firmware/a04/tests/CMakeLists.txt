cmake_minimum_required(VERSION 3.20.0)
project(aura_a04_codec_host C)
set(CMAKE_C_STANDARD 11)
# Zig exposes its archiver as a subcommand; CMake cannot discover a separate ar.
if(CMAKE_C_COMPILER MATCHES "zig(\\.exe)?$")
    set(CMAKE_AR "${CMAKE_C_COMPILER}")
    set(CMAKE_RANLIB "${CMAKE_C_COMPILER}")
    set(CMAKE_C_ARCHIVE_CREATE "<CMAKE_AR> ar qc <TARGET> <LINK_FLAGS> <OBJECTS>")
    set(CMAKE_C_ARCHIVE_APPEND "<CMAKE_AR> ar q <TARGET> <LINK_FLAGS> <OBJECTS>")
    set(CMAKE_C_ARCHIVE_FINISH "<CMAKE_RANLIB> ranlib <TARGET>")
endif()
include(../cmake/opus-profile.cmake)
add_executable(aura_codec_test host_codec.c ../src/aura_opus.c)
target_include_directories(aura_codec_test PRIVATE ../include)
target_compile_options(aura_codec_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(aura_codec_test PRIVATE opus)
